```rust
// ---------- season aggregates (redesign) ----------

#[derive(Debug, Clone)]
pub struct SeasonAgg {
    pub show_tmdb_id: i64,
    pub season_number: i64,
    pub total: i64,
    pub aired: i64,
    pub watched: i64,
    pub watched_aired: i64,
    pub dateless: i64,
    pub max_air: Option<String>,
    pub next_air: Option<String>,
    pub runtime_left_min: i64,
}

pub fn season_aggregates(conn: &Connection, today: &str) -> Result<Vec<SeasonAgg>> {
    let mut st = conn.prepare(
        "SELECT e.show_tmdb_id, e.season_number,
                COUNT(*),
                SUM(CASE WHEN e.air_date IS NOT NULL AND e.air_date<=?1 THEN 1 ELSE 0 END),
                SUM(CASE WHEN w.episode_tmdb_id IS NOT NULL THEN 1 ELSE 0 END),
                SUM(CASE WHEN w.episode_tmdb_id IS NOT NULL AND e.air_date IS NOT NULL AND e.air_date<=?1 THEN 1 ELSE 0 END),
                SUM(CASE WHEN e.air_date IS NULL THEN 1 ELSE 0 END),
                MAX(e.air_date),
                MIN(CASE WHEN e.air_date>?1 THEN e.air_date END),
                SUM(CASE WHEN e.air_date IS NOT NULL AND e.air_date<=?1 AND w.episode_tmdb_id IS NULL THEN COALESCE(e.runtime,0) ELSE 0 END)
         FROM episodes e
         LEFT JOIN episode_watches w ON w.episode_tmdb_id=e.tmdb_id
         WHERE e.season_number>0
         GROUP BY e.show_tmdb_id, e.season_number
         ORDER BY e.show_tmdb_id, e.season_number")?;
    let rows = st.query_map([today], |r| {
        Ok(SeasonAgg {
            show_tmdb_id: r.get(0)?, season_number: r.get(1)?, total: r.get(2)?,
            aired: r.get(3)?, watched: r.get(4)?, watched_aired: r.get(5)?,
            dateless: r.get(6)?, max_air: r.get(7)?, next_air: r.get(8)?,
            runtime_left_min: r.get(9)?,
        })
    })?;
    Ok(rows.collect::<std::result::Result<_, _>>()?)
}

#[derive(Debug, Clone)]
pub struct ShowBasics {
    pub tmdb_id: i64,
    pub name: String,
    pub poster_path: Option<String>,
    pub my_state: String,
    pub is_favorite: bool,
}

pub fn show_basics(conn: &Connection) -> Result<Vec<ShowBasics>> {
    let mut st = conn.prepare(
        "SELECT tmdb_id, name, poster_path, my_state, is_favorite FROM shows ORDER BY name")?;
    let rows = st.query_map([], |r| {
        Ok(ShowBasics {
            tmdb_id: r.get(0)?, name: r.get(1)?, poster_path: r.get(2)?,
            my_state: r.get(3)?, is_favorite: r.get::<_, i64>(4)? != 0,
        })
    })?;
    Ok(rows.collect::<std::result::Result<_, _>>()?)
}

pub fn last_activity_map(conn: &Connection) -> Result<HashMap<i64, String>> {
    let mut st = conn.prepare(
        "SELECT e.show_tmdb_id, MAX(w.watched_at) FROM episode_watches w
         JOIN episodes e ON e.tmdb_id=w.episode_tmdb_id
         WHERE w.watched_at IS NOT NULL
         GROUP BY e.show_tmdb_id")?;
    let mut map = HashMap::new();
    let rows = st.query_map([], |r| Ok((r.get::<_, i64>(0)?, r.get::<_, Option<String>>(1)?)))?;
    for row in rows {
        let (id, at) = row?;
        if let Some(at) = at {
            map.insert(id, at);
        }
    }
    Ok(map)
}
```

```rust
use std::collections::HashMap;

use rusqlite::Connection;

use crate::error::Result;
use crate::models::{AiringRow, AiringSeason, BingeCard, ReadySeason};
use crate::queries::{self, SeasonAgg};

pub struct ShowClass {
    pub ready: Option<ReadySeason>,
    pub airing: Option<AiringSeason>,
    pub series_watched_aired: i64,
    pub series_aired: i64,
    pub caught_up: bool,
}

fn is_complete(a: &SeasonAgg) -> bool {
    a.total > 0 && a.aired == a.total
}

/// Fold one show's season aggregates (season>0, ascending) into its binge state.
pub fn classify_show(aggs: &[SeasonAgg]) -> ShowClass {
    let show_watched: i64 = aggs.iter().map(|a| a.watched).sum();
    let series_aired: i64 = aggs.iter().map(|a| a.aired).sum();
    let series_watched_aired: i64 = aggs.iter().map(|a| a.watched_aired).sum();
    let ready = aggs.iter()
        .find(|a| is_complete(a) && a.watched < a.total)
        .map(|a| ReadySeason {
            season_number: a.season_number,
            ep_count: a.total,
            ep_left: a.total - a.watched,
            runtime_left_min: a.runtime_left_min,
            kind: if show_watched == 0 { "start" } else if a.watched > 0 { "finish" } else { "continue" }
                .to_string(),
        });
    let airing = aggs.iter()
        .find(|a| !is_complete(a) && (a.aired > 0 || a.next_air.is_some()))
        .map(|a| AiringSeason {
            season_number: a.season_number,
            aired_count: a.aired,
            total_count: a.total,
            next_air_date: a.next_air.clone(),
            finale_date: if a.dateless == 0 { a.max_air.clone() } else { None },
        });
    let caught_up = ready.is_none() && series_watched_aired >= series_aired;
    ShowClass { ready, airing, series_watched_aired, series_aired, caught_up }
}

pub fn build_binge_cards(conn: &Connection, today: &str) -> Result<Vec<BingeCard>> {
    let mut by_show: HashMap<i64, Vec<SeasonAgg>> = HashMap::new();
    for a in queries::season_aggregates(conn, today)? {
        by_show.entry(a.show_tmdb_id).or_default().push(a);
    }
    let activity = queries::last_activity_map(conn)?;
    let mut cards = Vec::new();
    for b in queries::show_basics(conn)? {
        let class = match by_show.get(&b.tmdb_id) {
            Some(aggs) => classify_show(aggs),
            None => ShowClass {
                ready: None, airing: None,
                series_watched_aired: 0, series_aired: 0, caught_up: true,
            },
        };
        cards.push(BingeCard {
            tmdb_id: b.tmdb_id,
            name: b.name,
            poster_path: b.poster_path,
            my_state: b.my_state,
            is_favorite: b.is_favorite,
            series_watched_aired: class.series_watched_aired,
            series_aired: class.series_aired,
            ready: class.ready,
            airing: class.airing,
            caught_up: class.caught_up,
            last_activity: activity.get(&b.tmdb_id).cloned(),
        });
    }
    Ok(cards)
}

/// Airing seasons of shows the user is watching, finale-known first (ascending), then name.
pub fn build_airing_rows(conn: &Connection, today: &str) -> Result<Vec<AiringRow>> {
    let mut rows: Vec<AiringRow> = build_binge_cards(conn, today)?
        .into_iter()
        .filter(|c| c.my_state == "watching")
        .filter_map(|c| {
            c.airing.map(|a| AiringRow {
                show_tmdb_id: c.tmdb_id,
                show_name: c.name,
                poster_path: c.poster_path,
                season_number: a.season_number,
                aired_count: a.aired_count,
                total_count: a.total_count,
                next_air_date: a.next_air_date,
                finale_date: a.finale_date,
            })
        })
        .collect();
    rows.sort_by(|x, y| match (&x.finale_date, &y.finale_date) {
        (Some(a), Some(b)) => a.cmp(b).then_with(|| x.show_name.cmp(&y.show_name)),
        (Some(_), None) => std::cmp::Ordering::Less,
        (None, Some(_)) => std::cmp::Ordering::Greater,
        (None, None) => x.show_name.cmp(&y.show_name),
    });
    Ok(rows)
```

```ts
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

/** "2026-08-24" -> "Aug 24" (adds the year when it isn't the current one). */
export function fmtDate(iso: string | null | undefined): string {
  if (!iso) return "";
  const [y, m, d] = iso.split("-").map(Number);
  if (!y || !m || !d) return iso;
  const year = new Date().getFullYear() === y ? "" : ` ${y}`;
  return `${MONTHS[m - 1]} ${d}${year}`;
}

/** 340 -> "5 h 40 m"; 45 -> "45 m"; 120 -> "2 h". */
export function fmtRuntime(min: number): string {
  if (min <= 0) return "";
  const h = Math.floor(min / 60);
  const m = min % 60;
  if (h === 0) return `${m} m`;
  if (m === 0) return `${h} h`;
  return `${h} h ${m} m`;
}

/** Shared label for an airing season (BingeCard.airing or AiringRow). */
export function airingLabel(a: {
  airedCount: number;
  nextAirDate: string | null;
  finaleDate: string | null;
}): string {
  if (a.airedCount === 0 && a.nextAirDate) return `starts ${fmtDate(a.nextAirDate)}`;
  if (a.finaleDate) return `bingeable ${fmtDate(a.finaleDate)}`;
  return "finale TBA";
}
```

```svelte
  /** Average color pushed to a vivid glow (sat >= .55, lightness .55). */
  function boost(r: number, g: number, b: number): [number, number, number] {
    const max = Math.max(r, g, b);
    const min = Math.min(r, g, b);
    const d = max - min;
    let h = 0;
    if (d !== 0) {
      if (max === r) h = ((g - b) / d) % 6;
      else if (max === g) h = (b - r) / d + 2;
      else h = (r - g) / d + 4;
      h *= 60;
      if (h < 0) h += 360;

    let s = max === 0 ? 0 : d / max;
    s = Math.max(s, 0.55);
    const l = 0.55;
    const c = (1 - Math.abs(2 * l - 1)) * s;
    const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
    const m = l - c / 2;
```

```ts
const MONTHS_FULL = [
  "January", "February", "March", "April", "May", "June",
  "July", "August", "September", "October", "November", "December",
];

/** "2026-08-24" -> "August 2026". */
export function fmtMonth(iso: string): string {
  const [y, m] = iso.split("-").map(Number);
  if (!y || !m) return iso;
  return `${MONTHS_FULL[m - 1]} ${y}`;
}
```

```svelte
    const aired = airedIn(eps);

    if (total > 0 && aired === total) return "ready";
    if (aired > 0 || eps.some((e) => e.airDate && e.airDate > today)) return "airing";
    return "tba";

    if (st === "airing") return `airing · ${airedIn(eps)}/${eps.length} aired`;
    if (st === "tba") return "dates TBA";
```
